async function collectMetrics() { const report = { timestamp: new Date().toISOString(), // Базовая инфо userAgent: navigator.userAgent, language: navigator.language, referrer: document.referrer || "Прямой переход", // Время и локализация timezone: Intl.DateTimeFormat().resolvedOptions().timeZone, timezoneOffset: new Date().getTimezoneOffset(), // Экран и интерфейс screen: { width: window.screen.width, height: window.screen.height, availWidth: window.screen.availWidth, availHeight: window.screen.availHeight, colorDepth: window.screen.colorDepth, pixelRatio: window.devicePixelRatio || 1 }, // Железо и ресурсы hardware: { cores: navigator.hardwareConcurrency || "N/A", ram: navigator.deviceMemory || "N/A", maxTouchPoints: navigator.maxTouchPoints || 0 // 0 = нет тачскрина }, // Конфиденциальность privacy: { doNotTrack: navigator.doNotTrack || "N/A", cookiesEnabled: navigator.cookieEnabled } }; // 1. Получаем данные о видеокарте через WebGL try { const canvas = document.createElement('canvas'); const gl = canvas.getContext('webgl') || canvas.getContext('experimental-webgl'); if (gl) { const debugInfo = gl.getExtension('WEBGL_debug_renderer_info'); report.graphics = { vendor: gl.getParameter(debugInfo.UNMASKED_VENDOR_WEBGL), renderer: gl.getParameter(debugInfo.UNMASKED_RENDERER_WEBGL) }; } else { report.graphics = { status: "WebGL не поддерживается" }; } } catch (e) { report.graphics = { status: "Заблокировано/Ошибка" }; } // 2. Получаем параметры сети (скорость, тип подключения) const conn = navigator.connection || navigator.mozConnection || navigator.webkitConnection; if (conn) { report.networkSpeed = { effectiveType: conn.effectiveType, // '4g', '3g', '2g' downlink: conn.downlink, // Скорость в Мбит/с rtt: conn.rtt // Задержка (ping) в мс }; } // 3. Получаем статус батареи (работает асинхронно) if (typeof navigator.getBattery === 'function') { try { const battery = await navigator.getBattery(); report.battery = { charging: battery.charging, level: Math.round(battery.level * 100) + '%' }; } catch (e) { report.battery = { status: "Доступ запрещен браузером" }; } } // 4. Запрос IP и Гео (внешний API) try { const response = await fetch('https://ipapi.co/json/'); if (response.ok) { const geo = await response.json(); report.geo = { ip: geo.ip, country: geo.country_name, region: geo.region, city: geo.city, asn: geo.asn, // Номер автономной системы (провайдера) org: geo.org }; } } catch (error) { report.geo = { error: "Не удалось получить IP" }; } // Вывод в консоль и на страницу console.log("Сформированный отпечаток устройства:", report); document.getElementById('output').textContent = JSON.stringify(report, null, 4); // Отправка (вызов твоей функции отправки) sendData(report); }
